Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0
Start-Service sshd
Set-Service sshd -StartupType Automatic


Get-Service sshd
Get-NetFirewallRule -Name OpenSSH-Server-In-TCP
ipconfig


New-NetFirewallRule `
  -Name "OpenSSH-Server-In-TCP" `
  -DisplayName "OpenSSH Server (SSH)" `
  -Enabled True `
  -Direction Inbound `
  -Protocol TCP `
  -LocalPort 22 `
  -Action Allow



New-Item -ItemType Directory -Force $HOME\.ssh
Get-Content $HOME\key.pub | Add-Content $HOME\.ssh\authorized_keys
Remove-Item $HOME\key.pub


Test-NetConnection 10.8.1.6 -Port 22
